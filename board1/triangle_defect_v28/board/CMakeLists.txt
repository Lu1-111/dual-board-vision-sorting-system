cmake_minimum_required(VERSION 3.15)
project(triangle_defect_npu LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs)
set(BUILD_APPS ON CACHE BOOL "" FORCE)
set(BUILD_MODULES OFF CACHE BOOL "" FORCE)
add_subdirectory("${FACE_DETECTION_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}/face_sdk")
add_executable(triangle_defect_npu main.cpp runtime_executor.cpp)
target_compile_options(triangle_defect_npu PRIVATE -O3)
target_include_directories(triangle_defect_npu PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(triangle_defect_npu PRIVATE flow_utils ${OpenCV_LIBS} pthread atomic dl)
target_link_options(triangle_defect_npu PRIVATE "-Wl,--as-needed")
add_executable(camera_probe camera_probe.cpp)
target_compile_options(camera_probe PRIVATE -O3)
target_include_directories(camera_probe PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(camera_probe PRIVATE ${OpenCV_LIBS})
add_executable(fb_snapshot fb_snapshot.cpp)
target_compile_options(fb_snapshot PRIVATE -O3)
target_include_directories(fb_snapshot PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(fb_snapshot PRIVATE ${OpenCV_LIBS})
